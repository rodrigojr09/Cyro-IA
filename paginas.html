export default async function Page({
  params,
  searchParams,
}: {
  params: Promise<{ section: string }>;
  searchParams: Promise<Record<string, string | string[] | undefined>>;
}) {
  const { section } = await params;
  if (
    ![
      "dashboard",
      "clients",
      "devices",
      "settings",
      "credentials",
      "versions",
      "usage",
      "audit",
    ].includes(section)
  )
    notFound();
  if (["credentials", "versions"].includes(section) && (await admin()).role !== "SUPER_ADMIN")
    notFound();
  const query = new URLSearchParams();
  if (section === "usage") {
    query.set("usage", "1");
    const filters = await searchParams;
    for (const key of ["from", "to", "clientId", "perUser", "usagePage"]) {
      if (typeof filters[key] === "string" && filters[key])
        query.set(key, filters[key]);
    }
  }
  return (
    <Dashboard
      section={section}
      query={query.size ? "?" + query.toString() : ""}
    />
  );
}

"use client";
import {
  createContext,
  useContext,
  useState,
  useSyncExternalStore,
  useEffect,
  useCallback,
} from "react";
import { createDashboardStore } from "../lib/dashboard-store";
const Context = createContext<ReturnType<typeof createDashboardStore> | null>(
  null,
);
export function DashboardDataProvider({
  children,
}: {
  children: React.ReactNode;
}) {
  const [store] = useState(() => createDashboardStore());
  return <Context.Provider value={store}>{children}</Context.Provider>;
}
export function useDashboardData<T>(key: string | null) {
  const store = useContext(Context);
  if (!store) throw Error("DashboardDataProvider ausente.");
  const resource = key || "/api/admin";
  const subscribe = useCallback(
    (listener: () => void) => store.subscribe(resource, listener),
    [store, resource],
  );
  const snapshot = useCallback(
    () => store.snapshot(resource),
    [store, resource],
  );
  const state = useSyncExternalStore(subscribe, snapshot, snapshot);
  useEffect(() => {
    if (key) void store.load(key);
  }, [store, key]);
  const reload = useCallback(
    () => (key ? store.load(key, true) : Promise.resolve()),
    [store, key],
  );
  return {
    data: key ? (state.data as T | null) : null,
    error: key ? state.error : "",
    loading: !!key && (state.loading || (!state.data && !state.error)),
    reload,
  };
}
export function DashboardLoading({
  error,
  retry,
}: {
  error?: string;
  retry?: () => void;
}) {
  return (
    <div
      className="dashboard-loading"
      role={error ? "alert" : "status"}
      aria-live="polite"
    >
      <div>
        {!error && <span className="loading-spinner" aria-hidden="true" />}
        <p>{error || "Carregando sua dashboard…"}</p>
        {error && <button onClick={retry}>Tentar novamente</button>}
      </div>
    </div>
  );
}

function Empty({ text }: { text: string }) {
  return (
    <div className="empty">
      <span>⌁</span>
      <p>{text}</p>
    </div>
  );
}
function Table({
  heads,
  rows,
}: {
  heads: string[];
  rows: React.ReactNode[][];
}) {
  return rows.length ? (
    <div
      className="table-wrap"
      role="region"
      aria-label={heads.join(", ")}
      tabIndex={0}
    >
      <table>
        <thead>
          <tr>
            {heads.map((h) => (
              <th scope="col" key={h}>
                {h}
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {rows.map((row, i) => (
            <tr key={i}>
              {row.map((cell, j) => (
                <td key={j}>{cell}</td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  ) : (
    <Empty text="Nenhum registro encontrado." />
  );
}

export function useAttachments() {
  const [files, setFiles] = useState([]),
    [choosing, setChoosing] = useState(false),
    [error, setError] = useState(''),
    epoch = useRef(0),
    selecting = useRef(false),
    filesRef = useRef(files);
  filesRef.current = files;
  const discard = (id) => void window.jarvisWindow.discardAttachment(id).catch(() => {});
  return {
    files,
    choosing,
    error,
    clear() {
      epoch.current++;
      filesRef.current.forEach((file) => discard(file.id));
      filesRef.current = [];
      setFiles([]);
      setError('');
    },
    remove(id) {
      const next = filesRef.current.filter((file) => file.id !== id);
      filesRef.current = next;
      setFiles(next);
      discard(id);
    },
    async paste(images, unavailable = '', source = 'clipboard') {
      if (unavailable) {
        setError(unavailable);
        return;
      }
      if (selecting.current) {
        setError('Aguarde a seleção de anexos terminar.');
        return;
      }
      if (!images.length) return;
      if (images.length + filesRef.current.length > 5) {
        setError('Limite de 5 arquivos por mensagem.');
        return;
      }
      if (
        images.some(
          (file) =>
            (source !== 'drop' &&
              !['image/png', 'image/jpeg', 'image/webp', 'image/gif'].includes(file.type)) ||
            !Number.isFinite(file.size) ||
            file.size <= 0 ||
            file.size > 5242880,
        )
      ) {
        setError(
          source === 'drop'
            ? 'Use arquivos compatíveis de até 5 MB.'
            : 'Use imagens PNG, JPEG, WebP ou GIF estático de até 5 MB.',
        );
        return;
      }
      selecting.current = true;
      setChoosing(true);
      setError('');
      const generation = epoch.current,
        created = [];
      try {
        for (const file of images) {
          const data = await new Promise((resolve, reject) => {
            const reader = new window.FileReader();
            reader.onload = () => resolve(String(reader.result).split(',')[1]);
            reader.onerror = () => reject(Error('Não foi possível ler a imagem colada.'));
            reader.readAsDataURL(file);
          });
          if (generation !== epoch.current) break;
          const result =
            source === 'drop'
              ? await window.jarvisWindow.dropFile({ data, name: file.name })
              : await window.jarvisWindow.pasteImage({ data, type: file.type });
          if (!result?.ok || !Array.isArray(result.data?.files))
            throw Error(result?.data?.detail || 'Não foi possível colar a imagem.');
          created.push(
            ...result.data.files.map((item) => ({ ...item, clipboard: source === 'clipboard' })),
          );
        }
        if (generation !== epoch.current) {
          created.forEach((file) => discard(file.id));
          return;
        }
        const next = [...filesRef.current, ...created];
        filesRef.current = next;
        setFiles(next);
        return true;
      } catch (error) {
        created.forEach((file) => discard(file.id));
        setError(error.message);
      } finally {
        selecting.current = false;
        setChoosing(false);
      }
    },
    async choose() {
      if (selecting.current || files.length >= 5) return;
      selecting.current = true;
      setChoosing(true);
      setError('');
      const generation = epoch.current;
      try {
        const result = await window.jarvisWindow.chooseAttachments(5 - files.length);
        if (result.canceled) return;
        if (!result.ok) throw Error(result.data?.detail || 'Não foi possível anexar.');
        const selected = result.data.files;
        if (!Array.isArray(selected))
          throw Error('Não foi possível conferir os anexos selecionados.');
        if (generation !== epoch.current) {
          selected.forEach((file) => discard(file.id));
          return;
        }
        if (
          selected.some((file) => !Number.isFinite(file.size) || file.size > 5242880) ||
          files.length + selected.length > 5
        )
          throw Error('Limite de 5 arquivos de até 5 MB.');
        const next = [...filesRef.current, ...selected];
        filesRef.current = next;
        setFiles(next);
      } catch (error) {
        setError(error.message);
      } finally {
        selecting.current = false;
        setChoosing(false);
      }
    },
  };

// Measured outside the sidebar so row titles are never used as the popup surface.
export function AnchoredMenu({ menu, onClose, children }) {
  const ref = useRef(null);
  const [position, setPosition] = useState(null);
  useLayoutEffect(() => {
    const node = ref.current;
    const bounds = node.getBoundingClientRect();
    const anchor = menu.anchor;
    const gap = 8;
    const width = window.innerWidth;
    const height = window.innerHeight;
    let left = anchor.right + gap;
    let top = anchor.top;
    if (menu.placement === 'below-end') {
      left = Math.max(gap, Math.min(anchor.right - bounds.width, width - bounds.width - gap));
      top = anchor.bottom + gap;
      if (top + bounds.height > height - gap) top = anchor.top - bounds.height - gap;
    }
    if (menu.placement !== 'below-end' && left + bounds.width > width - gap) {
      if (anchor.left - bounds.width - gap >= gap) left = anchor.left - bounds.width - gap;
      else {
        left = Math.max(gap, width - bounds.width - gap);
        top = anchor.bottom + gap;
        if (top + bounds.height > height - gap) top = anchor.top - bounds.height - gap;
      }
    }
    setPosition({
      left: Math.max(gap, left),
      top: Math.max(gap, Math.min(top, height - bounds.height - gap)),
    });
    const close = () => onClose();
    const scroll = (event) => {
      if (
        !node.contains(event.target) &&
        (event.target === document || event.target.contains?.(menu.trigger))
      )
        close();
    };
    const outside = (event) => {
      if (!node.contains(event.target) && !menu.trigger?.contains(event.target)) close();
    };
    window.addEventListener('resize', close);
    document.addEventListener('scroll', scroll, true);
    document.addEventListener('pointerdown', outside);
    return () => {
      window.removeEventListener('resize', close);
      document.removeEventListener('scroll', scroll, true);
      document.removeEventListener('pointerdown', outside);
    };
  }, [menu]);
  useLayoutEffect(() => {
    if (position)
      ref.current
        ?.querySelector('button:not(:disabled):not([hidden])')
        ?.focus({ preventScroll: true });
  }, [position]);
  return createPortal(
    <div
      ref={ref}
      role="menu"
      aria-label={menu.label || (menu.kind === 'create' ? 'Criar no projeto' : 'Ações da conversa')}
      className={
        'ui-action-menu workspace-action-menu ' +
        (menu.kind === 'create' ? 'project-create-menu' : 'context-menu')
      }
      style={{ ...position, visibility: position ? 'visible' : 'hidden' }}
      onKeyDown={(event) => {
        if (event.key === 'Escape' || event.key === 'Tab') {
          if (event.key === 'Escape') event.preventDefault();
          onClose();
          menu.trigger?.focus({ preventScroll: true });
        } else if (['ArrowDown', 'ArrowUp', 'Home', 'End'].includes(event.key)) {
          event.preventDefault();
          const items = [...ref.current.querySelectorAll('button:not(:disabled):not([hidden])')];
          const index = items.indexOf(document.activeElement);
          const next =
            event.key === 'Home'
              ? 0
              : event.key === 'End'
                ? items.length - 1
                : (index + (event.key === 'ArrowUp' ? -1 : 1) + items.length) % items.length;
          items[next]?.focus({ preventScroll: true });
        }
      }}
    >
      {children}
    </div>,
    document.body,
  );
}

export function Markdown({ text = '' }) {
  const nodes = useMemo(() => {
    const html = DOMPurify.sanitize(marked.parse(String(text), { gfm: true, breaks: true }), {
      ALLOWED_TAGS: tags,
      ALLOWED_ATTR: ['href', 'title', 'class'],
      ALLOW_DATA_ATTR: false,
    });
    const document = new DOMParser().parseFromString(html, 'text/html');
    const convert = (node, key) => {
      if (node.nodeType === 3) return node.textContent;
      const tag = node.tagName?.toLowerCase();
      if (!tags.includes(tag)) return null;
      const children = [...node.childNodes]
        .filter(
          (item) =>
            !['table', 'thead', 'tbody', 'tr'].includes(tag) ||
            item.nodeType !== 3 ||
            item.textContent.trim(),
        )
        .map((item, index) => convert(item, index));
      if (tag === 'br' || tag === 'hr') return React.createElement(tag, { key });
      if (tag === 'a') {
        const href = node.getAttribute('href');
        return (
          <a
            key={key}
            href={/^https?:\/\//i.test(href || '') ? href : undefined}
            onClick={(event) => {
              event.preventDefault();
              if (/^https?:\/\//i.test(href || ''))
                void window.jarvisWindow.openLink(href).catch(() => {});
            }}
          >
            {children}
          </a>
        );
      }
      if (tag === 'pre')
        return (
          <Code
            key={key}
            text={node.textContent || ''}
            language={node.querySelector('code')?.className.match(/language-([\w+-]+)/)?.[1] || ''}
          />
        );
      return React.createElement(tag, { key }, children);
    };
    return [...document.body.childNodes].map(convert);
  }, [text]);
  return <div className="markdown">{nodes}</div>;
}

function RichEditor({ writer, document: documentValue, readOnly, onRun }) {
  const connected = useRef(null),
    applying = useRef(false),
    initial = useRef(null);
  if (!initial.current)
    initial.current = { extensions: extensions(), content: toDocument(documentValue.blocks) };
  const editor = useEditor(
    {
      extensions: initial.current.extensions,
      content: initial.current.content,
      shouldRerenderOnTransaction: false,
      editorProps: {
        attributes: {
          class: 'cyro-document-editor',
          role: 'textbox',
          'aria-label': 'Conteúdo do documento',
          'aria-multiline': 'true',
          spellcheck: 'true',
        },
      },
      onUpdate: ({ editor }) => {
        if (!applying.current && !connected.current?.adapter.remote)
          writer.edit(editorBlocks(editor));
      },
      onSelectionUpdate: ({ editor }) => {
        const { from, to, empty } = editor.state.selection;
        if (empty) {
          writer.select([], '');
          return;
        }
        const ranges = [];
        editor.state.doc.forEach((node, pos) => {
          if (from >= pos + node.nodeSize || to <= pos || node.isLeaf) return;
          const left = Math.max(0, from - pos - 1),
            right = Math.min(node.content.size, to - pos - 1);
          if (right < left) return;
          const start = Array.from(node.textBetween(0, left, '\n', '\n')).length;
          const end = Array.from(node.textBetween(0, right, '\n', '\n')).length;
          const text = Array.from(nodeText(node.toJSON())).slice(start, end).join('');
          if (text) ranges.push({ block_id: node.attrs.id, start, end, text });
        });
        writer.select(
          ranges.map((r) => r.block_id),
          editor.state.doc.textBetween(from, to, '\n', '\n'),
          ranges.length === 1 ? ranges[0] : ranges,
        );
      },
    },
    [],
  );
  useEffect(() => {
    if (!editor) return;
    connected.current = connectEditor(editor, writer);
    const preserve = () => writer.preserve();
    window.addEventListener('beforeunload', preserve);
    return () => {
      preserve();
      connected.current?.dispose();
      connected.current = null;
      window.removeEventListener('beforeunload', preserve);
    };
  }, [editor, writer]);
  useEffect(() => {
    editor?.setEditable(!readOnly, false);
  }, [editor, readOnly]);
  useEffect(() => {
    if (editor && connected.current) {
      applying.current = true;
      try {
        connected.current.adapter.apply(documentValue.blocks);
      } finally {
        applying.current = false;
      }
    }
  }, [editor, documentValue]);
  if (!editor) return null;
  const run = (command) => {
    if (!readOnly) editor.chain().focus()[command]().run();
  };
  return (
    <>
      <div
        className="writer-formatting"
        aria-label="Formatação do documento"
        onMouseDown={(event) => {
          if (event.target.closest('button')) event.preventDefault();
        }}
      >
        <select
          aria-label="Estilo de parágrafo"
          defaultValue="paragraph"
          disabled={readOnly}
          onChange={(e) => {
            const value = e.target.value;
            if (value === 'paragraph') editor.chain().focus().setParagraph().run();
            else if (value === 'bullet') editor.chain().focus().toggleBulletList().run();
            else if (value === 'number') editor.chain().focus().toggleOrderedList().run();
            else
              editor
                .chain()
                .focus()
                .toggleHeading({ level: Number(value) })
                .run();
          }}
        >
          <option value="paragraph">Texto normal</option>
          {[1, 2, 3].map((level) => (
            <option key={level} value={level}>
              Título {level}
            </option>
          ))}
          <option value="bullet">Lista com marcadores</option>
          <option value="number">Lista numerada</option>
        </select>
        {[
          ['toggleBold', 'Negrito', 'B'],
          ['toggleItalic', 'Itálico', 'I'],
          ['toggleUnderline', 'Sublinhado', 'U'],
        ].map(([command, label, text]) => (
          <button key={command} aria-label={label} disabled={readOnly} onClick={() => run(command)}>
            {text}
          </button>
        ))}
        <select
          aria-label="Alinhamento"
          defaultValue="left"
          disabled={readOnly}
          onChange={(e) => editor.chain().focus().setTextAlign(e.target.value).run()}
        >
          {[
            ['left', 'Esquerda'],
            ['center', 'Centro'],
            ['right', 'Direita'],
            ['justify', 'Justificado'],
          ].map(([value, text]) => (
            <option key={value} value={value}>
              {text}
            </option>
          ))}
        </select>
        <button
          aria-label="Inserir ou remover link"
          disabled={readOnly}
          onClick={() =>
            onRun(async () => {
              const href = await onRun.ask(
                'Endereço do link',
                editor.getAttributes('link').href || '',
                'Use HTTPS, HTTP ou mailto. Deixe vazio para remover.',
              );
              if (href === null) return;
              if (!href) {
                editor.chain().focus().extendMarkRange('link').unsetLink().run();
                return;
              }
              const url = new URL(href);
              if (
                !['https:', 'http:', 'mailto:'].includes(url.protocol) ||
                url.username ||
                url.password
              )
                throw Error('Endereço inválido.');
              editor.chain().focus().setLink({ href }).run();
            })
          }
        >
          Link
        </button>
        <details>
          <summary>Inserir</summary>
          <button
            disabled={readOnly}
            onClick={() =>
              editor.chain().focus().insertTable({ rows: 3, cols: 3, withHeaderRow: true }).run()
            }
          >
            Tabela
          </button>
          <button
            disabled={readOnly}
            onClick={() => editor.chain().focus().insertContent({ type: 'pageBreak' }).run()}
          >
            Quebra de página
          </button>
        </details>
      </div>
      <div id="writer-paper">
        <EditorContent editor={editor} />
      </div>
    </>
  );
}
export function WriterProposal({ writer, onRun }) {
  const state = useSyncExternalStore(writer.subscribe, writer.get),
    proposal = state.proposal;
  if (!proposal) return null;
  const pending = proposal.state === 'PENDING';
  return (
    <section id="writer-proposal" aria-label="Revisão proposta">
      <strong>Proposta do Cyro</strong>
      <p id="writer-countdown" role="status">
        {proposal.state === 'CONFLICTED'
          ? 'Conflito: o documento mudou. Peça uma nova proposta.'
          : 'O documento só será alterado após sua aprovação.'}
      </p>
      <div className="writer-proposal-text">
        <Markdown
          text={
            proposal.plan?.blocks?.map((b) => b.text).join('\n\n') || proposal.plan?.operation || ''
          }
        />
      </div>
      <button
        id="writer-approve"
        disabled={!pending || state.busy}
        onClick={() => onRun(() => writer.decide(true))}
      >
        Aprovar alteração
      </button>
      <button
        id="writer-reject"
        disabled={!pending || state.busy}
        onClick={() => onRun(() => writer.decide(false))}
      >
        Descartar proposta
      </button>
    </section>
  );
}
export function WriterDivider() {
  const drag = useRef(false);
  const adjust = (element, width) => {
    const root = element.parentElement;
    root.style.setProperty(
      '--writer-chat-width',
      Math.max(280, Math.min(root.clientWidth * 0.6, width)) + 'px',
    );
  };
  return (
    <div
      className="writer-divider"
      role="separator"
      aria-label="Redimensionar chat lateral"
      aria-orientation="vertical"
      tabIndex={0}
      onPointerDown={(e) => {
        drag.current = true;
        e.currentTarget.setPointerCapture(e.pointerId);
      }}
      onPointerMove={(e) => {
        if (drag.current)
          adjust(
            e.currentTarget,
            e.currentTarget.parentElement.getBoundingClientRect().right - e.clientX,
          );
      }}
      onPointerUp={() => {
        drag.current = false;
      }}
      onKeyDown={(e) => {
        if (['ArrowLeft', 'ArrowRight'].includes(e.key)) {
          e.preventDefault();
          adjust(
            e.currentTarget,
            e.currentTarget.parentElement.querySelector('.writer-standard-chat').clientWidth +
              (e.key === 'ArrowLeft' ? 24 : -24),
          );
        }
      }}
    />
  );
}
